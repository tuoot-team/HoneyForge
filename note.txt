Удалить агента: 
dest=/opt/honeyforge
if [ "$(id -u)" -ne 0 ]; then
	dest="$HOME/.honeyforge"
fi
if [ -f "$dest/agent.pid" ]; then
	kill "$(cat "$dest/agent.pid")" 2>/dev/null || true
fi
for d in /proc/[0-9]*; do
	if [ "$(readlink "$d/exe" 2>/dev/null)" = "$dest/agent" ]; then
		kill "${d#/proc/}" 2>/dev/null || true
	fi
done
rm -rf "$dest"
